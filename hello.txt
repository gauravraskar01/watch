<script>
  const t=document.getElementById('time'),d=document.getElementById('date'),f=document.getElementById('fill');
  function tick(){
    const n=new Date(),p=x=>String(x).padStart(2,'0');
    t.innerHTML=p(n.getHours())+':'+p(n.getMinutes())+'<small id="sec"> '+p(n.getSeconds())+'</small>';
    d.textContent=n.toLocaleDateString('en',{weekday:'short',day:'numeric',month:'short'});
    f.style.width=(n.getSeconds()/60*100)+'%';
  }
  tick();setInterval(tick,1000);
